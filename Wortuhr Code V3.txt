/* ========== Helpers: avoid touching WIFI_LEDS when drawing words ========== */

inline bool isWifiLed(uint8_t idx) {
  for (uint8_t i = 0; i < 4; i++)
    if (WIFI_LEDS[i] == idx) return true;
  return false;
}

/* ========== Target drawing helpers (do not write WIFI_LEDS) ========== */

void clearTarget() {
  fill_solid(targetLeds, NUM_LEDS, CRGB::Black);
}

void setWordTarget(const uint8_t *arr, uint8_t len) {
  for (uint8_t i = 0; i < len; i++) {
    uint8_t idx = arr[i];
    if (idx < NUM_LEDS && !isWifiLed(idx)) targetLeds[idx] = WORD_COLOR;
  }
}

void setMinuteDotsTarget(uint8_t count) {
  for (uint8_t i = 0; i < 4; i++) {
    uint8_t idx = MIN_LEDS[i];
    if (idx < NUM_LEDS) targetLeds[idx] = (i < count) ? WORD_COLOR : CRGB::Black;
  }
}

/* ========== Crossfade: blend current leds[] -> targetLeds[] (only changed pixels appear to move) ========== */

void crossFade(uint8_t steps = 45, uint16_t delayMs = 18) {
  for (uint8_t s = 0; s <= steps; s++) {
    uint8_t amt = (255 * s) / steps;
    for (int i = 0; i < NUM_LEDS; i++) {
      leds[i] = blend(leds[i], targetLeds[i], amt);
    }
    FastLED.show();
    delay(delayMs);
  }
}

/* ========== Time -> target builder ========== */

void showHourTarget(uint8_t h) {
  switch (h) {
    case 1: setWordTarget(EINS, 4); break;
    case 2: setWordTarget(ZWEI, 4); break;
    case 3: setWordTarget(DREI, 4); break;
    case 4: setWordTarget(VIER_H, 4); break;
    case 5: setWordTarget(FUENF_H, 4); break;
    case 6: setWordTarget(SECHS, 5); break;
    case 7: setWordTarget(SIEBEN, 6); break;
    case 8: setWordTarget(ACHT, 4); break;
    case 9: setWordTarget(NEUN, 4); break;
    case 10: setWordTarget(ZEHN_H, 4); break;
    case 11: setWordTarget(ELF, 3); break;
    case 12: setWordTarget(ZWOELF, 5); break;
  }
}

/* determine whether 'hour' falls into night mode given nightStart/nightEnd (handles wrap around midnight) */
bool isNightHour(uint8_t hour) {
  if (nightStartHour == nightEndHour) return false;  // degenerate: no night
  if (nightStartHour < nightEndHour) {
    return (hour >= nightStartHour && hour < nightEndHour);
  } else {
    // e.g., 22 .. 6
    return (hour >= nightStartHour || hour < nightEndHour);
  }
}

void buildTimeTarget(int hourVal, int minuteVal) {

  // set brightness according to night/day percentage
  uint8_t effectivePercent = isNightHour(hourVal) ? nightBrightPercent : dayBrightPercent;
  uint8_t brightnessVal = map(effectivePercent, 0, 100, 0, 255);
  FastLED.setBrightness(brightnessVal);

  clearTarget();
  setWordTarget(ES_IST, 5);

  uint8_t rounded = (minuteVal / 5) * 5;
  uint8_t extra = minuteVal % 5;

  uint8_t h = hourVal % 12;
  if (h == 0) h = 12;
  // For minutes >=25 the spoken hour is the next hour for both dialects
  if (rounded >= 25) h = (h % 12) + 1;

  // Dialect-specific logic:
  if (dialectMode == 1) {
    // Ost-Logik: 15 -> "Viertel" + next hour (no "nach")
    //            45 -> "Dreiviertel" + next hour
    switch (rounded) {
      case 0:
        if (h == 1) {
          setWordTarget(EIN, 3);  // EIN UHR
        } else {
          showHourTarget(h);
        }
        setWordTarget(UHR, 3);
        break;
      case 5:
        setWordTarget(FUENF, 4);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 10:
        setWordTarget(ZEHN, 4);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 15:
        {
          uint8_t nextHour = (h % 12) + 1;
          setWordTarget(VIERTEL, 7);
          showHourTarget(nextHour);
        }
        break;
        break;  // Viertel + next hour
      case 20:
        setWordTarget(ZWANZIG, 7);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 25:
        setWordTarget(FUENF, 4);
        setWordTarget(VOR, 3);
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 30:
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 35:
        setWordTarget(FUENF, 4);
        setWordTarget(NACH, 4);
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 40:
        setWordTarget(ZWANZIG, 7);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
      case 45:
        setWordTarget(DREIVIERTEL, 11);
        showHourTarget(h);
        break;  // Dreiviertel + next hour
      case 50:
        setWordTarget(ZEHN, 4);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
      case 55:
        setWordTarget(FUENF, 4);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
    }
  } else {
    // West-Logik
    switch (rounded) {
      case 0:
        if (h == 1) {
          setWordTarget(EIN, 3);  // EIN UHR
        } else {
          showHourTarget(h);
        }
        setWordTarget(UHR, 3);
        break;
      case 5:
        setWordTarget(FUENF, 4);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 10:
        setWordTarget(ZEHN, 4);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 15:
        setWordTarget(VIERTEL, 7);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 20:
        setWordTarget(ZWANZIG, 7);
        setWordTarget(NACH, 4);
        showHourTarget(h);
        break;
      case 25:
        setWordTarget(FUENF, 4);
        setWordTarget(VOR, 3);
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 30:
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 35:
        setWordTarget(FUENF, 4);
        setWordTarget(NACH, 4);
        setWordTarget(HALB, 4);
        showHourTarget(h);
        break;
      case 40:
        setWordTarget(ZWANZIG, 7);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
      case 45:
        setWordTarget(VIERTEL, 7);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
      case 50:
        setWordTarget(ZEHN, 4);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
      case 55:
        setWordTarget(FUENF, 4);
        setWordTarget(VOR, 3);
        showHourTarget(h);
        break;
    }
  }

  setMinuteDotsTarget(extra);
  // NOTE: WIFI_LEDS remain untouched here
}

void handleSave() {
  // params:
  // c (color #RRGGBB), day, night, ns, ne, di, y, mo, da, h, mi
  if (server.hasArg("c")) {
    String c = server.arg("c");
    if (c.length() == 7 && c[0] == '#') {
      long v = strtol(c.substring(1).c_str(), NULL, 16);
      WORD_COLOR = CRGB((v >> 16) & 255, (v >> 8) & 255, v & 255);
    }
  }
  if (server.hasArg("day")) {
    int d = server.arg("day").toInt();
    if (d < 0) d = 0;
    if (d > 100) d = 100;
    dayBrightPercent = d;
  }
  if (server.hasArg("night")) {
    int n = server.arg("night").toInt();
    if (n < 0) n = 0;
    if (n > 100) n = 100;
    nightBrightPercent = n;
  }
  if (server.hasArg("ns")) {
    int ns = server.arg("ns").toInt();
    if (ns < 0) ns = 0;
    if (ns > 23) ns = 0;
    nightStartHour = ns;
  }
  if (server.hasArg("ne")) {
    int ne = server.arg("ne").toInt();
    if (ne < 0) ne = 0;
    if (ne > 23) ne = 0;
    nightEndHour = ne;
  }
  if (server.hasArg("di")) {
    int di = server.arg("di").toInt();
    if (di != 0 && di != 1) di = 0;
    dialectMode = di;
  }

  // Date/time
  bool gotDate = false;
  int y = 0, mo = 0, da = 0, hh = 0, mm = 0;
  if (server.hasArg("y") && server.hasArg("mo") && server.hasArg("da") && server.hasArg("h") && server.hasArg("mi")) {
    y = server.arg("y").toInt();
    mo = server.arg("mo").toInt();
    da = server.arg("da").toInt();
    hh = server.arg("h").toInt();
    mm = server.arg("mi").toInt();
    // basic validation
    if (y >= 2000 && y <= 2099 && mo >= 1 && mo <= 12 && da >= 1 && da <= 31 && hh >= 0 && hh <= 23 && mm >= 0 && mm <= 59) {
      gotDate = true;
    } else {
      gotDate = false;
    }
  }

  // Save to Preferences
  prefs.begin("wortuhr", false);
  prefs.putUChar(P_DB, dayBrightPercent);
  prefs.putUChar(P_NB, nightBrightPercent);
  prefs.putUChar(P_NS, nightStartHour);
  prefs.putUChar(P_NE, nightEndHour);
  prefs.putUChar(P_R, WORD_COLOR.r);
  prefs.putUChar(P_G, WORD_COLOR.g);
  prefs.putUChar(P_B, WORD_COLOR.b);
  prefs.putUChar(P_DI, dialectMode);
  prefs.end();

  // If date provided, set RTC
  if (gotDate) {
    rtc.adjust(DateTime(y, mo, da, hh, mm, 0));
  }

  server.send(200, "text/html",
              "<html><body style='background:#111;color:white;text-align:center;font-family:Arial'>"
              "<h2>Gespeichert!</h2><p>Uhr startet neu...</p></body></html>");

  delay(900);
  dnsServer.stop();
  WiFi.softAPdisconnect(true);
  delay(200);
  ESP.restart();
}

// Neu: Handler zum Setzen der RTC per UTC-Epoch Sekunden
void handleSetTime() {

  if (server.hasArg("y") && server.hasArg("mo") && server.hasArg("da") && server.hasArg("h") && server.hasArg("mi") && server.hasArg("s")) {

    int y = server.arg("y").toInt();
    int mo = server.arg("mo").toInt();
    int da = server.arg("da").toInt();
    int h = server.arg("h").toInt();
    int mi = server.arg("mi").toInt();
    int s = server.arg("s").toInt();

    rtc.adjust(DateTime(y, mo, da, h, mi, s));
    server.send(200, "text/plain", "OK");
    return;
  }

  server.send(400, "text/plain", "BAD");
}


void handleNotFound() {
  // redirect everything to root (captive portal)
  IPAddress apIP = WiFi.softAPIP();
  server.sendHeader("Location", String("http://") + apIP.toString() + "/", true);
  server.send(302, "text/plain", "");
}

/* ========== Preferences load ========== */
void loadSettings() {
  prefs.begin("wortuhr", true);
  dayBrightPercent = prefs.getUChar(P_DB, DEFAULT_DAY_BRI_PERCENT);
  nightBrightPercent = prefs.getUChar(P_NB, DEFAULT_NIGHT_BRI_PERCENT);
  nightStartHour = prefs.getUChar(P_NS, DEFAULT_NIGHT_START);
  nightEndHour = prefs.getUChar(P_NE, DEFAULT_NIGHT_END);
  uint8_t r = prefs.getUChar(P_R, DEFAULT_COLOR.r);
  uint8_t g = prefs.getUChar(P_G, DEFAULT_COLOR.g);
  uint8_t b = prefs.getUChar(P_B, DEFAULT_COLOR.b);
  WORD_COLOR = CRGB(r, g, b);
  dialectMode = prefs.getUChar(P_DI, DEFAULT_DIALECT);
  prefs.end();
}

/* ========== Start AP & DNS & Webserver ========== */
void startAP() {
  WiFi.mode(WIFI_AP);
  // open AP (no password) to simplify setup
  WiFi.softAP("WORTUHR-SETUP");

  apStartTime = millis();
  apActive = true;
  clientConnected = false;

  // DNS: capture everything and redirect to softAP IP
  IPAddress apIP = WiFi.softAPIP();
  dnsServer.start(DNS_PORT, "*", apIP);

  // webserver handlers
  server.on("/", handleRoot);
  server.on("/generate_204", handleGenerate204);  // android captive portal
  server.on("/fwlink", handleGenerate204);
  server.on("/save", handleSave);
  server.on("/settime", handleSetTime);  // neu: Endpoint um RTC per /settime?ts=.. zu setzen
  server.onNotFound(handleNotFound);
  server.begin();
}

/* ========== Update status LEDs 37-40 (apActive / clientConnected) ========== */
void applyWifiStatusVisuals() {
  if (apActive) {
    int stations = WiFi.softAPgetStationNum();
    if (stations > 0) {
      // device connected -> solid blue
      for (uint8_t i = 0; i < 4; i++) {
        uint8_t idx = WIFI_LEDS[i];
        if (idx < NUM_LEDS) leds[idx] = CRGB(0, 120, 255);
      }
      return;
    } else {
      // AP active, no client -> pulsing blue
      pulseValue += pulseDir;
      if (pulseValue >= 200 || pulseValue <= 6) pulseDir = -pulseDir;
      uint8_t pv = constrain(pulseValue, 0, 255);
      for (uint8_t i = 0; i < 4; i++) {
        uint8_t idx = WIFI_LEDS[i];
        if (idx < NUM_LEDS) leds[idx] = CRGB(0, 0, pv);
      }
      return;
    }
  } else {
    // AP disabled -> ensure off
    for (uint8_t i = 0; i < 4; i++) {
      uint8_t idx = WIFI_LEDS[i];
      if (idx < NUM_LEDS) leds[idx] = CRGB::Black;
    }
  }
}

/* ========== Core time show wrapper ========== */
int lastMinute = -1;

void showTimeAndFade(int h, int m) {
  buildTimeTarget(h, m);  // create targetLeds (does NOT touch WIFI_LEDS)
  crossFade();            // animate current leds -> targetLeds
  // After crossFade, target is already visible in leds[]
  // But status LEDs (37-40) must be applied immediately afterward if AP active
  applyWifiStatusVisuals();
  FastLED.show();
}

void startAnimation() {

  FastLED.setBrightness(120);  // feste Starthelligkeit

  // alles aus
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();

  // Reihenfolge mischen
  uint8_t order[NUM_LEDS];
  for (int i = 0; i < NUM_LEDS; i++) order[i] = i;

  for (int i = NUM_LEDS - 1; i > 0; i--) {
    int j = random(i + 1);
    uint8_t t = order[i];
    order[i] = order[j];
    order[j] = t;
  }

  // LEDs zufällig bunt einschalten
  for (int i = 0; i < NUM_LEDS; i++) {
    leds[order[i]] = CHSV(random8(), 255, 200);
    FastLED.show();
    delay(35);  // bestimmt Gesamtdauer
  }

  delay(500);

  // weich ausblenden
  for (int b = 200; b >= 0; b -= 5) {
    FastLED.setBrightness(b);
    FastLED.show();
    delay(20);
  }

  FastLED.setBrightness(255);
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();
}


/* ========== Setup & loop ========== */

void setup() {
  Serial.begin(115200);
  delay(50);

  FastLED.addLeds<WS2812, DATA_PIN, GRB>(leds, NUM_LEDS);
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  fill_solid(targetLeds, NUM_LEDS, CRGB::Black);

  Wire.begin();
  rtc.begin();

  if (rtc.lostPower()) {
    // fallback: set compile time
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }

  loadSettings();

  startAnimation();

  // Always start AP initially
  startAP();

  // Immediately show current time (small crossfade)
  DateTime now = rtc.now();
  lastMinute = now.minute();
  buildTimeTarget(now.hour(), now.minute());
  crossFade(18, 12);

  // After the initial crossfade, overlay wifi visuals
  applyWifiStatusVisuals();
  FastLED.show();
}

void loop() {
  // handle DNS and captive portal
  dnsServer.processNextRequest();
  server.handleClient();

  // Track ap active timeout: if no client connects within 60s, shut down AP
  if (apActive) {
    int stations = WiFi.softAPgetStationNum();
    if (stations > 0) {
      clientConnected = true;  // note a client connected
    }

    if (!clientConnected && (millis() - apStartTime > 60000UL)) {
      // shutdown AP, no client connected in timeout
      dnsServer.stop();
      WiFi.softAPdisconnect(true);
      apActive = false;
      clientConnected = false;
      // make sure wifi leds off
      for (uint8_t i = 0; i < 4; i++) {
        if (WIFI_LEDS[i] < NUM_LEDS) leds[WIFI_LEDS[i]] = CRGB::Black;
      }
      FastLED.show();
    }

    // If a client had connected and then left (stations == 0),
    // stop AP as soon as it disconnects (follow original 8266 behaviour)
    if (clientConnected && (WiFi.softAPgetStationNum() == 0) && (millis() - apStartTime > 500)) {
      dnsServer.stop();
      WiFi.softAPdisconnect(true);
      apActive = false;
      clientConnected = false;
      for (uint8_t i = 0; i < 4; i++) {
        if (WIFI_LEDS[i] < NUM_LEDS) leds[WIFI_LEDS[i]] = CRGB::Black;
      }
      FastLED.show();
    }
  }

  // Re-apply wifi status visuals to leds array (pulsing or solid) continuously
  applyWifiStatusVisuals();

  // Only update time when minute changes
  DateTime now = rtc.now();
  if (now.minute() != lastMinute) {
    lastMinute = now.minute();
    showTimeAndFade(now.hour(), now.minute());
  } else {
    // still show latest leds (with status overlay)
    FastLED.show();
  }

  // small delay keeps loop responsive & smooth pulses
  delay(80);
}
